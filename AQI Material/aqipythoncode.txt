import numpy as np
import pandas as pd
from datetime import datetime

# Step 1: Load the dataset
df = pd.read_csv('aqi.csv')

# Step 2: Display a few records of the original dataset
print("Few records of original dataset:\n", df.head())

# Step 3: Check for missing values
print("Missing values in each column:\n", df.isnull().sum())

# Step 4: Drop duplicate rows
df.drop_duplicates(inplace=True)

# Step 5: Replacing all instances of 0 with np.nan
df.replace(0, np.nan, inplace=True)

# Step 6: Fill missing values in numerical columns with the median of each column, rounded to 2 decimal places
df.fillna(df.select_dtypes(include=[np.number]).median().round(2), inplace=True)

# Step 7: Handling Outliers using IQR for numerical columns
Q1 = df.select_dtypes(include=[np.number]).quantile(0.25)
Q3 = df.select_dtypes(include=[np.number]).quantile(0.75)
IQR = Q3 - Q1

# Defining the lower and upper bounds for outliers
lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

# Filter out outliers from numerical columns only
df = df[~((df.select_dtypes(include=[np.number]) < lower_bound) |
          (df.select_dtypes(include=[np.number]) > upper_bound)).any(axis=1)]

# Step 8: Handling Data Inconsistencies (example: removing negative values or converting incorrect data types)
df = df[df['aqi'] > 0]  # Ensure AQI values are positive
df['pm2.5'] = pd.to_numeric(df['pm2.5'], errors='coerce')  # Convert PM2.5 to numeric, invalid entries become NaN

# Step 9: Handling date format inconsistencies
# Define a function to convert the 'Date' column using a custom format
def parse_date(date_str):
    try:
        # Attempt parsing using multiple common formats
        return pd.to_datetime(date_str, format='%d-%m-%Y', errors='coerce') or pd.to_datetime(date_str, format='%m-%d-%Y', errors='coerce')
    except Exception as e:
        return pd.NaT  # Return NaT for any invalid date formats

# Apply the custom date parsing function
df['date'] = df['date'].apply(parse_date)

# Step 10: Display the cleaned dataset
print("Cleaned Dataset:\n", df)

# Step 11: Save the cleaned dataset to a new CSV file
df.to_csv('cleaned_with_aqi.csv', index=False)
